```js
 /**
     * 设置锚地点类型高亮颜色
     * @param entity
     */
    setAnchorageLight(entity) {
      this.handleClearAnchorLight();
      this.viewer.entities.add({
        id: 'anchorageCursor',
        hidePopup: true,
        position: entity.position,
        billboard: {
          image: shipCursor,
          width: 15,
          height: 15,
          scaleByDistance: new Cesium.NearFarScalar(1000, 1, 10000, 1.5),
          // pixelOffset: new Cesium.Cartesian2(0, -20),
        },
      });
    },
```
