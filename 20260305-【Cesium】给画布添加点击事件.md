             _this.tpInvestPointClick(entity);
             return;
           }
           if (entity.otherType === 2 || entity.otherType === 3) {
             // 点击视频
             let lnglat = entity.longitude + ',' + entity.latitude;

             const areaId = topics[topics.length - 2];

                 code: r.description,

             _this.drawShipCursor(entity);
           } else if (entity.gisType === 'shipTrackPoint') {

             console.log('轨迹点');
           } else if (entity.gisType === 'heatMap') {
             // 热力图点
             _this.queryHeatData(movement);
           } else if (
             entity.gisType === 1 &&
             entity.entityType === 1 &&
             entity.otherType === 0 && // 是否为风电场，这个判断的可靠性存疑
             _this.tpModuleShow && // 打开台风开关
             _this.tpCurrList?.length // 当前有台风

               actionId: 'click_wind_farm_icon',
               actionName: '点击风电场图标',

           } else if (entity.gisType === 1 && entity.name === '锚地' && _this.tpModuleShow) { // 台风模式下点击锚地显示弹窗，其他模式下还是鼠标移入显示
             window.webTracker.sendEvent({
               actionId: 'click_anchor_icon',
               actionName: '点击锚地图标',

             _this.setAnchorageLight(entity);

           } else if (entity.gisType === 3 && entity.name === '锚地' && _this.tpModuleShow) { // 点击面类型的锚地
             window.webTracker.sendEvent({

             // 设置多边形点击高亮颜色

_this.viewer.screenSpaceEventHandler.setInputAction 是 Cesium 里**绑定屏幕交互事件**的方法。
它用于给地图画布注册鼠标/触摸/键盘组合事件的回调。

常见用法：

```js
viewer.screenSpaceEventHandler.setInputAction((movement) => {
  // 点击后执行
}, Cesium.ScreenSpaceEventType.LEFT_CLICK)
```

核心点：

1. `screenSpaceEventHandler`：监听画布上的输入事件（点、拖、移动等）。

2. `setInputAction(callback, eventType, modifier?)`：给某种事件绑定处理函数。

3. `eventType`

   常见值：

   - `LEFT_CLICK`
   - `LEFT_DOUBLE_CLICK`
   - `MOUSE_MOVE`
   - `RIGHT_CLICK`
   - `LEFT_DOWN / LEFT_UP`
   - `WHEEL` 等

如果要取消绑定，用：

```js
viewer.screenSpaceEventHandler.removeInputAction(Cesium.ScreenSpaceEventType.LEFT_CLICK)
```
